nes.h:
/* Tiny NES API for the test ROM and the Claude scenes (MIT).
 *
 * OAM shadow: 64 entries x 4 bytes at $0200 (Y, tile, attributes, X), copied
 * to the PPU with OAM DMA in every NMI. A sprite with OAM Y = y is drawn on
 * lines y+1 .. y+8 (y+16 in 8x16 mode). Y = $EF..$FF hides it.
 */
#ifndef NES_H
#define NES_H

#define OAM ((unsigned char *)0x0200)
#define OAM_Y(i)    OAM[((i) << 2) + 0]
#define OAM_TILE(i) OAM[((i) << 2) + 1]
#define OAM_ATTR(i) OAM[((i) << 2) + 2]
#define OAM_X(i)    OAM[((i) << 2) + 3]

#define NMI_COUNT (*(volatile unsigned char *)0x07F1)
#define DEBUG0    (*(volatile unsigned char *)0x07F2)   /* free for the program */
#define TAG       (*(volatile unsigned char *)0x07F3)   /* copied to SHOWN_TAG at each OAM DMA */
#define OAM_READY (*(volatile unsigned char *)0x07F4)
#define SHOWN_TAG (*(volatile unsigned char *)0x07F5)
#define LAG       (*(volatile unsigned char *)0x07F6)   /* NMIs that found no finished table */

/* shadows written to PPUCTRL / PPUMASK in the NMI */
extern unsigned char ppu_ctrl, ppu_mask;
#pragma zpsym ("ppu_ctrl")
#pragma zpsym ("ppu_mask")
#define CTRL_SPR_8X16 0x20

#define PAD_A      0x80
#define PAD_B      0x40
#define PAD_SELECT 0x20
#define PAD_START  0x10
#define PAD_UP     0x08
#define PAD_DOWN   0x04
#define PAD_LEFT   0x02
#define PAD_RIGHT  0x01

void wait_frame(void);                    /* wait for the next NMI */
void oam_ready(void);                     /* the OAM shadow is complete: copy it in the next NMI */
unsigned char pad_read(void);             /* controller 1 */
void ppu_setup(const unsigned char *pal); /* rendering off, 32-byte palette, blank nametable, rendering on */
void oam_hide_all(void);                  /* every byte of OAM to $FF */
unsigned char rnd8(void);                 /* 8-bit xorshift */

#endif


main.c:
/* main.c - Rain: 40 falling drops (8x8) and a d-pad-controlled umbrella (32x16).
 *
 * Sprite budget: the umbrella takes OAM slots 0-7 (the highest priority) so
 * it never loses sprites to the 8-per-scanline limit. The 40 drops take
 * slots 8-47. Their priority order rotates every frame.
 *
 * The umbrella uses 4 sprites on each of lines UMB_Y..UMB_Y+15, so at most
 * 4 drops may overlap any one of those lines. A per-line counter (band_cnt)
 * tracks this; a drop whose next step would put a 5th drop on a line in the
 * band simply waits a frame, so the hardware limit is never exceeded.
 */
#include "nes.h"

#define NUM_DROPS   40
#define DROP_TILE   0x08   /* small dot */
#define UMB_Y       200    /* OAM Y of the canopy's top row */
#define UMB_H       16
#define UMB_W       32
#define UMB_SPEED   3
#define DROP_BOTTOM 232
#define BAND_MAX    4      /* drops allowed per line inside the umbrella band */

static const unsigned char palette[32] = {
    /* background */
    0x0F, 0x0F, 0x0F, 0x0F,  0x0F, 0x0F, 0x0F, 0x0F,
    0x0F, 0x0F, 0x0F, 0x0F,  0x0F, 0x0F, 0x0F, 0x0F,
    /* sprites: pal 0 = rain, pal 1 = umbrella */
    0x0F, 0x21, 0x11, 0x30,
    0x0F, 0x16, 0x06, 0x38,
    0x0F, 0x21, 0x11, 0x30,
    0x0F, 0x21, 0x11, 0x30
};

static unsigned char dx[NUM_DROPS];
static unsigned char dy[NUM_DROPS];
static unsigned char dsp[NUM_DROPS];
static unsigned char band_cnt[UMB_H];

static unsigned char ux;
static unsigned char rot;

/* add (+1) or remove (-1) a drop at row y from the band counters */
static void band_add(unsigned char y, signed char d)
{
    unsigned char l, hi;
    if (y + 8 <= UMB_Y || y >= UMB_Y + UMB_H) return;
    l = (y > UMB_Y) ? y : UMB_Y;
    hi = y + 7;
    if (hi > UMB_Y + UMB_H - 1) hi = UMB_Y + UMB_H - 1;
    for (; l <= hi; ++l) band_cnt[l - UMB_Y] += d;
}

/* can a drop at row y be added without exceeding BAND_MAX on any line? */
static unsigned char band_fits(unsigned char y)
{
    unsigned char l, hi;
    if (y + 8 <= UMB_Y || y >= UMB_Y + UMB_H) return 1;
    l = (y > UMB_Y) ? y : UMB_Y;
    hi = y + 7;
    if (hi > UMB_Y + UMB_H - 1) hi = UMB_Y + UMB_H - 1;
    for (; l <= hi; ++l) {
        if (band_cnt[l - UMB_Y] >= BAND_MAX) return 0;
    }
    return 1;
}

static void respawn(unsigned char i)
{
    unsigned char x = rnd8();
    if (x > 247) x -= 8;
    dx[i] = x;
    dy[i] = 0;
}

static void draw_umbrella(void)
{
    unsigned char c, row, n = 0;
    for (row = 0; row < 2; ++row) {
        for (c = 0; c < 4; ++c) {
            OAM_Y(n)    = UMB_Y + (row << 3);
            OAM_TILE(n) = row ? 0x02 : 0x01;
            OAM_ATTR(n) = 1;
            OAM_X(n)    = ux + (c << 3);
            ++n;
        }
    }
}

void main(void)
{
    unsigned char i, k, pad, slot, y, t;
    unsigned int  top;

    ppu_setup(palette);
    oam_hide_all();

    ux = (256 - UMB_W) / 2;
    rot = 0;

    for (i = 0; i < UMB_H; ++i) band_cnt[i] = 0;

    for (i = 0; i < NUM_DROPS; ++i) {
        dx[i] = rnd8();
        if (dx[i] > 247) dx[i] -= 8;
        t = rnd8() % DROP_BOTTOM;
        while (!band_fits(t)) t = rnd8() % DROP_BOTTOM;
        dy[i] = t;
        band_add(t, 1);
        dsp[i] = 1 + (i & 3) + ((i >> 2) & 1);   /* speeds 1..5 */
    }

    for (;;) {
        /* --- umbrella input --- */
        pad = pad_read();
        if (pad & PAD_LEFT) {
            ux = (ux > UMB_SPEED) ? ux - UMB_SPEED : 0;
        }
        if (pad & PAD_RIGHT) {
            ux += UMB_SPEED;
            if (ux > 256 - UMB_W) ux = 256 - UMB_W;
        }

        /* --- drops --- */
        for (i = 0; i < NUM_DROPS; ++i) {
            y = dy[i] + dsp[i];
            top = y;
            band_add(dy[i], -1);
            if (y >= DROP_BOTTOM) {
                respawn(i);
            } else if (top + 8 > UMB_Y && top < UMB_Y + 16 &&
                       (unsigned int)dx[i] + 8 > ux &&
                       (unsigned int)dx[i] < (unsigned int)ux + UMB_W) {
                respawn(i);          /* caught by the umbrella */
            } else if (band_fits(y)) {
                dy[i] = y;
            }
            /* else: wait a frame so no line gets a 5th drop in the band */
            band_add(dy[i], 1);
        }

        /* --- OAM --- */
        draw_umbrella();
        for (k = 0; k < NUM_DROPS; ++k) {
            i = k + rot;
            if (i >= NUM_DROPS) i -= NUM_DROPS;
            slot = 8 + k;
            OAM_Y(slot)    = dy[i];
            OAM_TILE(slot) = DROP_TILE;
            OAM_ATTR(slot) = 0;
            OAM_X(slot)    = dx[i];
        }
        ++rot;
        if (rot >= NUM_DROPS) rot = 0;

        oam_ready();
        wait_frame();
    }
}
